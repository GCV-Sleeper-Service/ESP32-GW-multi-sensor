## 5. Exact Scope — Network Card Renderer

Add `CARD_RENDERERS.network` to the dashboard to render the WAN ping probe as a network card.

### 5a. Architecture decision: SENSORS filtering approach

**Use Option A: Include network devices in SENSORS.** This is the cleaner approach and matches the architecture plan. However, this requires careful handling of `makeSensorConfig()` which is ThermoPro-shaped.

Modify `normalizeManifestSensors()` to include all categories:

```js
function normalizeManifestSensors(payload) {
  var sensors = [];
  if (Array.isArray(payload)) sensors = payload;
  else if (payload && Array.isArray(payload.sensors)) sensors = payload.sensors;
  return sensors.map(function(sensor) {
    return {
      id: String(sensor && sensor.id || '').trim(),
      name: String(sensor && sensor.name || '').trim(),
      category: String(sensor && sensor.category || 'environmental').trim(),
      metrics: Array.isArray(sensor && sensor.metrics) ? sensor.metrics : []
    };
  }).filter(function(sensor) {
    return sensor.id && sensor.name;
    // Removed environmental-only filter — all categories now included
  });
}
```

### 5b. ⚠️ CRITICAL: Create `makeNetworkSensorConfig()` for non-ThermoPro devices

The existing `makeSensorConfig()` generates ThermoPro-specific entity IDs like `text_sensor-{id}_temperature`. These IDs map to ESPHome text_sensor entities published by the YAML lambdas. The ping device does NOT have ESPHome text_sensor entities — its data comes through `/api/v2/live`, not SSE state events.

Add a category-aware sensor config factory:

```js
function makeNetworkSensorConfig(meta, idx) {
  var id = meta.id, name = meta.name;
  var color = SENSOR_COLORS[idx % SENSOR_COLORS.length];
  try { var saved = localStorage.getItem('sensor_color_' + id); if (saved) color = saved; } catch(e) {}
  return {
    id: id, name: name,
    color: color,
    category: 'network',
    // No ThermoPro entity IDs — network devices use /api/v2/live polling
    restPaths: []
  };
}
```

Update `applySensorMeta()` to dispatch by category:

```js
function applySensorMeta(meta) {
  if (!Array.isArray(meta) || !meta.length) meta = DEFAULT_SENSOR_META;
  App.State.setSensors(meta.map(function(m, idx) {
    var cat = m.category || 'environmental';
    if (cat === 'network') return makeNetworkSensorConfig(m, idx);
    if (cat === 'system') return makeNetworkSensorConfig(m, idx); // reuse for now
    return makeSensorConfig(m, idx); // ThermoPro path
  }));
  try { App.Features.emit('onManifest', App.State.getSensors()); } catch(e) { logNonFatal('manifest hook emit', e); }
}
```

### 5c. Implement `buildNetworkCard(device, manifest)` function

```js
function buildNetworkCard(s, manifest) {
  // Find source.target from manifest for display
  var target = '—';
  if (manifest && manifest.sensors) {
    var entry = manifest.sensors.find(function(ms) { return ms.id === s.id; });
    if (entry && entry.source && entry.source.target) target = entry.source.target;
  }
  return (
    '<div class="sensor-card network-card" data-device-id="' + s.id + '">' +
      '<div class="sensor-card-header">' +
        '<input class="sensor-color-picker" id="picker-' + s.id + '" type="color" value="' + (s.color || '#4FC3F7') + '" data-sensor-color="' + s.id + '">' +
        s.name +
      '</div>' +
      '<div class="sensor-readings">' +
        '<div class="sensor-reading">' +
          '<div class="reading-label">Latency</div>' +
          '<div class="reading-value waiting" id="net-ping-' + s.id + '" style="color:' + (s.color || '#4FC3F7') + '">&mdash;</div>' +
        '</div>' +
        '<div class="sensor-reading">' +
          '<div class="reading-label">Success Rate</div>' +
          '<div class="reading-value waiting" id="net-success-' + s.id + '">&mdash;</div>' +
        '</div>' +
        '<div class="sensor-reading">' +
          '<div class="reading-label">Target</div>' +
          '<div class="reading-value" id="net-target-' + s.id + '">' + target + '</div>' +
        '</div>' +
        '<div class="sensor-reading">' +
          '<div class="reading-label">Last Seen</div>' +
          '<div class="reading-time" id="net-lastseen-' + s.id + '">last: &mdash;</div>' +
        '</div>' +
      '</div>' +
    '</div>'
  );
}
```

Register it:
```js
CARD_RENDERERS.network = function(device, manifest) {
  return buildNetworkCard(device, manifest);
};
```

### 5e. ⚠️ CRITICAL: Network device data path — `/api/v2/live` polling

The ping device data does NOT come through ESPHome SSE state events. The `handleState()` function processes SSE events by matching ThermoPro entity IDs (`s.tempId`, `s.humId`, etc.). The ping device has no such entities.

Instead, the network card gets live data from the `/api/v2/live` endpoint. You need to add a periodic polling function that:

1. Fetches `/api/v2/live` on the same interval as the existing polling cycle (or a dedicated cycle)
2. Extracts network device values from the response
3. Updates network card DOM elements

Add a `updateNetworkCards()` function:

```js
function updateNetworkCards(liveData) {
  // liveData = response from /api/v2/live → { timestamp, devices: { wan_ping: { ping_ms, success_pct, last_seen } } }
  if (!liveData || !liveData.devices) return;
  SENSORS.forEach(function(s) {
    if (s.category !== 'network') return;
    var devData = liveData.devices[s.id];
    if (!devData) return;

    var pingEl = document.getElementById('net-ping-' + s.id);
    if (pingEl && devData.ping_ms !== undefined && devData.ping_ms !== null) {
      pingEl.textContent = METRIC_FORMATTERS.ping_latency(devData.ping_ms);
      pingEl.classList.remove('waiting');
    }

    var successEl = document.getElementById('net-success-' + s.id);
    if (successEl && devData.success_pct !== undefined && devData.success_pct !== null) {
      successEl.textContent = METRIC_FORMATTERS.success_rate(devData.success_pct);
      successEl.classList.remove('waiting');
    }

    var seenEl = document.getElementById('net-lastseen-' + s.id);
    if (seenEl && devData.last_seen) {
      var d = new Date(devData.last_seen * 1000);
      seenEl.textContent = 'last: ' + d.toLocaleTimeString([], {hour12:false});
    }
  });
}
```

Then add a periodic `/api/v2/live` fetch, e.g. every 15 seconds, in the boot flow or alongside the existing SSE/polling mechanism. Use an in-flight guard (LESSON-OPS-050):

```js
var _v2LiveInFlight = false;
function pollV2Live() {
  if (_v2LiveInFlight) return;
  _v2LiveInFlight = true;
  fetch(ESP_HOST + '/api/v2/live', {cache:'no-store'})
    .then(function(r) { if (!r.ok) throw new Error('HTTP ' + r.status); return r.json(); })
    .then(function(data) { updateNetworkCards(data); })
    .catch(function(err) { dlog('v2/live poll: ' + err.message, 'err'); })
    .finally(function() { _v2LiveInFlight = false; });
}
```

Start polling after boot:
```js
// In boot flow, after manifest loaded and cards built:
setInterval(pollV2Live, 15000);
pollV2Live(); // initial fetch
```

### 5f. Chart rendering for network devices

The existing `fetchDeviceHistory()` is manifest-driven and uses `history_url` from the v2 manifest. The wan_ping manifest entry has `history_url: "/api/v2/history/wan_ping/ping_ms"` and `"/api/v2/history/wan_ping/success_pct"`. The generic history fetcher should work for network devices IF the chart infrastructure can handle non-temperature/humidity datasets.

For this step, **do NOT add chart rendering for network devices.** The chart code is tightly coupled to temperature/humidity datasets with hardcoded chart IDs (`tempChart`, `humChart`). Adding network charts requires a separate Chart.js instance or a generic chart registry, which is a larger change.

Instead, ensure that `buildDeviceCards()` skips chart-related initialization for network devices. The network card should display live values only (no chart canvas).
