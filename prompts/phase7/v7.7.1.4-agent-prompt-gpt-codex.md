### Task Group 1: Add retention budget calculator to `firmware/core/nvs-persistence.h`

Add after the per-device persist functions (from v7.7.1.3):

```cpp
// ── Phase 7: Retention budget calculator (v7.7.1.4) ────────────
// Calculates per-device max_slots based on partition size, device count,
// and priority tiers. See Docs/v7.7-v7.8-persistence-architecture.md §7.
//
// Priority tiers:
//   high (70%)   — environmental sensors (long-term trending)
//   normal (20%) — moderate retention (soil, leak, distance)
//   low (10%)    — operational (ping, host health)
//
// Devices without history_enabled metrics get 0 slots.

static constexpr int PER_SEGMENT_NVS_OVERHEAD = 48;   // NVS entry header + alignment
static constexpr int PER_META_NVS_OVERHEAD = 48;
static constexpr float NVS_USABLE_FRACTION = 0.85f;   // page overhead, alignment waste

struct RetentionBudget {
  int max_slots;           // calculated slot count for this device
  const char *device_id;   // which device this budget is for
};

static int calculate_retention_budgets_(RetentionBudget *budgets, int max_budgets) {
  // Determine partition size — use the live 3-arg signature:
  // find_partition_size_bytes_(label, type, subtype)
  // PERFORMANCE-ACK: calculate_retention_budgets_() MUST remain off the HTTP request path.
  // Allowed call sites for this helper:
  //   (1) restore_device_history_v2_() — boot path (once per boot / after reset)
  //   (2) persist_device_segment_() — persist path (once per device per cycle; expected ~hourly)
  //
  // Complexity notes:
  //   - Boot path: O(device_count) NVS/array scan work.
  //   - Persist path: O(device_count) per call × N devices per cycle ⇒ O(N²) per persist cycle.
  //
  // Do NOT justify this choice with guessed timing numbers. If performance is a concern, measure
  // the real device and either (a) cache budgets per cycle or (b) compute once at boot and reuse
  // until device_count changes.
  uint32_t partition_bytes = find_partition_size_bytes_(HISTORY_PARTITION_LABEL,
                                                        ESP_PARTITION_TYPE_DATA,
                                                        ESP_PARTITION_SUBTYPE_DATA_NVS);
  if (partition_bytes == 0) {
    ESP_LOGE(TAG, "Retention budget: history partition not found");
    return 0;
  }

  float usable_bytes = partition_bytes * NVS_USABLE_FRACTION;

  // Count persistent devices and classify by priority tier
  int high_count = 0, normal_count = 0, low_count = 0;
  int persistent_count = 0;
  int per_slot_cost = sizeof(DeviceSegment) + PER_SEGMENT_NVS_OVERHEAD;
  int per_meta_cost = sizeof(DeviceHistoryMeta) + PER_META_NVS_OVERHEAD;

  for (int d = 0; d < NUM_DEVICES && d < max_budgets; d++) {
    budgets[d].device_id = devices[d].id;
    budgets[d].max_slots = 0;

    // Check if device has any history-enabled metrics
    bool has_history = false;
    for (int m = 0; m < devices[d].metric_count; m++) {
      if (devices[d].metric_defs[m].history_enabled &&
          devices[d].metric_states[m].history != nullptr) {
        has_history = true;
        break;
      }
    }
    if (!has_history) continue;

    persistent_count++;

    // Classify by category: environmental=high, network=low, system=low
    switch (devices[d].category_id) {
      case 0: high_count++; break;    // environmental
      case 1: low_count++; break;     // system
      case 2: low_count++; break;     // network
      default: normal_count++; break;
    }
  }

  if (persistent_count == 0) return 0;

  // Subtract meta overhead from usable bytes
  float segment_budget = usable_bytes - (persistent_count * per_meta_cost);
  // Also subtract space used by old engine meta + segments (coexistence)
  // Conservative: reserve 30% for old engine during dual-write phase
  segment_budget *= 0.70f;

  if (segment_budget <= 0) {
    ESP_LOGW(TAG, "Retention budget: insufficient space after meta reservation");
    return persistent_count;
  }

  // Calculate tier budgets with redistribution
  float high_share = (high_count > 0) ? 0.70f : 0.0f;
  float normal_share = (normal_count > 0) ? 0.20f : 0.0f;
  float low_share = (low_count > 0) ? 0.10f : 0.0f;
  float total_share = high_share + normal_share + low_share;

  // Redistribute unused tiers
  if (total_share > 0 && total_share < 1.0f) {
    float scale = 1.0f / total_share;
    high_share *= scale;
    normal_share *= scale;
    low_share *= scale;
  }

  int high_slots = (high_count > 0) ? (int)((segment_budget * high_share) / (high_count * per_slot_cost)) : 0;
  int normal_slots = (normal_count > 0) ? (int)((segment_budget * normal_share) / (normal_count * per_slot_cost)) : 0;
  int low_slots = (low_count > 0) ? (int)((segment_budget * low_share) / (low_count * per_slot_cost)) : 0;

  // Cap at 999 (3-digit key suffix limit)
  if (high_slots > 999) high_slots = 999;
  if (normal_slots > 999) normal_slots = 999;
  if (low_slots > 999) low_slots = 999;

  // Assign budgets
  for (int d = 0; d < NUM_DEVICES && d < max_budgets; d++) {
    bool has_history = false;
    for (int m = 0; m < devices[d].metric_count; m++) {
      if (devices[d].metric_defs[m].history_enabled &&
          devices[d].metric_states[m].history != nullptr) {
        has_history = true;
        break;
      }
    }
    if (!has_history) continue;

    switch (devices[d].category_id) {
      case 0: budgets[d].max_slots = high_slots; break;
      case 1: budgets[d].max_slots = low_slots; break;
      case 2: budgets[d].max_slots = low_slots; break;
      default: budgets[d].max_slots = normal_slots; break;
    }

    ESP_LOGI(TAG, "Retention budget: %s → %d slots (%.1f days)",
             devices[d].id, budgets[d].max_slots,
             budgets[d].max_slots / 24.0f);
  }

  return persistent_count;
}
```

### Task Group 2: Add v2 restore function to `firmware/core/nvs-persistence.h`

Add after the retention budget calculator:

```cpp
// ── Phase 7: Per-device restore from NVS (v7.7.1.4) ────────────
// Reads per-device segments from NVS at boot, populating RAM history
// buffers. Runs alongside old restore_from_nvs() (dual-restore) until
// v7.7.2.1 switchover.

static bool restore_device_history_v2_() {
  nvs_handle_t handle;
  if (!open_history_nvs_(&handle, NVS_READWRITE)) {
    ESP_LOGE(TAG, "V2 restore: failed to open NVS");
    return false;
  }

  // Calculate retention budgets
  RetentionBudget budgets[NUM_DEVICES] = {};
  int persistent_count = calculate_retention_budgets_(budgets, NUM_DEVICES);

  if (persistent_count == 0) {
    ESP_LOGI(TAG, "V2 restore: no persistent devices");
    nvs_close(handle);
    return true;
  }

  int restored_devices = 0;
  for (int d = 0; d < NUM_DEVICES; d++) {
    if (budgets[d].max_slots == 0) continue;

    maybe_yield_nvs_scan_(d);  // Rule 61

    // Load device meta
    DeviceHistoryMeta meta = {};
    bool have_meta = load_device_meta_(handle, devices[d].id, &meta);
    if (!have_meta || meta.valid_segments == 0) {
      ESP_LOGD(TAG, "V2 restore: no data for %s", devices[d].id);
      continue;
    }

    // Verify device_id matches (collision detection)
    if (strncmp(meta.device_id, devices[d].id, sizeof(meta.device_id)) != 0) {
      ESP_LOGW(TAG, "V2 restore: hash collision for %s (meta says %s)",
               devices[d].id, meta.device_id);
      continue;
    }

    // Update max_slots if budget changed
    if (meta.max_slots != (uint16_t)budgets[d].max_slots) {
      ESP_LOGI(TAG, "V2 restore: %s budget changed %d → %d slots",
               devices[d].id, meta.max_slots, budgets[d].max_slots);
      meta.max_slots = budgets[d].max_slots;
      // Cap valid_segments if budget shrank
      if (meta.valid_segments > meta.max_slots) {
        meta.valid_segments = meta.max_slots;
      }
      save_device_meta_(handle, devices[d].id, meta);
    }

    // Read segments in chronological order (oldest first)
    int oldest_slot = (meta.next_slot + meta.max_slots - meta.valid_segments) % meta.max_slots;
    int segments_loaded = 0;

    for (int n = 0; n < meta.valid_segments; n++) {
      maybe_yield_nvs_scan_(n);  // Rule 61

      int slot = (oldest_slot + n) % meta.max_slots;
      char key[16];
      make_device_segment_key_(devices[d].id, slot, key, sizeof(key));

      DeviceSegment segment = {};
      size_t len = sizeof(DeviceSegment);
      esp_err_t err = nvs_get_blob(handle, key, &segment, &len);
      if (err != ESP_OK || len != sizeof(DeviceSegment)) continue;

      if (segment.header.magic != DEV_HIST_MAGIC ||
          segment.header.version != DEV_HIST_VERSION) continue;

      // Populate RAM history buffers from segment data
      int persist_idx = 0;
      for (int m = 0; m < devices[d].metric_count && m < MAX_PERSIST_METRICS; m++) {
        if (!devices[d].metric_defs[m].history_enabled ||
            devices[d].metric_states[m].history == nullptr) continue;

        HistoryBuffer *buf = devices[d].metric_states[m].history;
        int count = segment.counts[persist_idx];
        for (int p = 0; p < count && p < PERSIST_POINTS_PER_SEGMENT; p++) {
          const HistEntry &entry = segment.data[persist_idx][p];
          if (entry.epoch > 0) buf->add(entry.epoch, entry.value);
        }
        persist_idx++;
      }
      segments_loaded++;
    }

    if (segments_loaded > 0) {
      ESP_LOGI(TAG, "V2 restore: %s — %d/%d segments loaded",
               devices[d].id, segments_loaded, meta.valid_segments);
      restored_devices++;
    }
  }

  nvs_close(handle);
  ESP_LOGI(TAG, "V2 restore complete: %d/%d devices with data",
           restored_devices, persistent_count);
  return true;
}
```
