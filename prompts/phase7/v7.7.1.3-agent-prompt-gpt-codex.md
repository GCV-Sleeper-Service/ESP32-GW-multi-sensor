```cpp

// ═══════════════════════════════════════════════════════════════════
// Phase 7 — per-device persist write path (v7.7.1.3, PLAN-046)
//   Each hourly persist also writes one DeviceSegment per persistent
//   device (dual-write). Nothing reads these blobs until v7.7.1.4.
// ═══════════════════════════════════════════════════════════════════

// The ring of each device in this step: one day of hourly segments. v7.7.1.4
// replaces it by the retention budget (PLAN-046: the 360 slots of May 2026 do not fit).
static constexpr uint16_t DEV_PERSIST_PROVISIONAL_SLOTS = 24;
// A v2 write needs this many free NVS entries in the history partition, so that the
// old engine keeps room for its hourly overwrite: two pages of 126 entries.
static constexpr uint32_t DEV_PERSIST_MIN_FREE_ENTRIES = 252;

// The result of the last persist cycle of one device.
enum DevPersistResult : uint8_t {
  DEV_PERSIST_PENDING = 0,    // no cycle since the boot
  DEV_PERSIST_OK = 1,         // the segment and the meta are written
  DEV_PERSIST_DEDUP = 2,      // the newest point is already persisted
  DEV_PERSIST_EMPTY = 3,      // the device has no point to persist
  DEV_PERSIST_NVS_LOW = 4,    // fewer free entries than DEV_PERSIST_MIN_FREE_ENTRIES
  DEV_PERSIST_MISSING = 5,    // the id is not in devices[]
  DEV_PERSIST_COLLISION = 6,  // the meta blob of the key holds another device id
  DEV_PERSIST_ERROR = 7,      // an allocation, a metric mismatch or an NVS call failed
};

static constexpr const char *dev_persist_result_name_(uint8_t result) {
  return result == DEV_PERSIST_PENDING ? "pending"
       : result == DEV_PERSIST_OK ? "ok"
       : result == DEV_PERSIST_DEDUP ? "dedup"
       : result == DEV_PERSIST_EMPTY ? "empty"
       : result == DEV_PERSIST_NVS_LOW ? "nvs_low"
       : result == DEV_PERSIST_MISSING ? "missing"
       : result == DEV_PERSIST_COLLISION ? "collision"
       : result == DEV_PERSIST_ERROR ? "error" : "unknown";
}

// RAM only: the persist path writes it, /api/storage-stats reads it, on two tasks. Both take a copy of
// one device's status through the two functions below, never the array itself (PLAN-047).
struct DevicePersistStatus {
  uint32_t last_persist_epoch = 0;  // from the device's meta after the cycle
  uint32_t writes_ok = 0;           // results OK since the boot
  uint32_t writes_skipped = 0;      // results DEDUP, EMPTY and NVS_LOW since the boot
  uint32_t writes_failed = 0;       // results MISSING, COLLISION and ERROR since the boot
  uint16_t valid_segments = 0;      // from the device's meta after the cycle
  uint16_t next_slot = 0;           // from the device's meta after the cycle
  uint8_t last_result = DEV_PERSIST_PENDING;
};
static DevicePersistStatus g_dev_persist_status[PERSIST_DEVICE_ARRAY_LEN] = {};
static portMUX_TYPE g_dev_persist_status_mux = portMUX_INITIALIZER_UNLOCKED;

// A copy of the status of one persistent device, taken under a short critical section.
static DevicePersistStatus dev_persist_status_get_(int persist_index) {
  DevicePersistStatus status;
  if (persist_index < 0 || persist_index >= PERSIST_DEVICE_COUNT) return status;
  portENTER_CRITICAL(&g_dev_persist_status_mux);
  status = g_dev_persist_status[persist_index];
  portEXIT_CRITICAL(&g_dev_persist_status_mux);
  return status;
}

// Publishes the status of one persistent device, under the same critical section.
static void dev_persist_status_put_(int persist_index, const DevicePersistStatus &status) {
  if (persist_index < 0 || persist_index >= PERSIST_DEVICE_COUNT) return;
  portENTER_CRITICAL(&g_dev_persist_status_mux);
  g_dev_persist_status[persist_index] = status;
  portEXIT_CRITICAL(&g_dev_persist_status_mux);
}
```

**2.2** After S2, at the end of the file, write these eight definitions yourself, with your file-edit tool, in this order. Their interface is fixed: write the first line of each exactly as the block below shows it, without its `;` and its comment, followed by ` {` on the same line:

```cpp
static constexpr uint16_t dev_ring_next_(uint16_t slot, uint16_t max_slots);                 // the slot after slot
static constexpr uint16_t dev_ring_valid_(uint16_t valid, uint16_t max_slots);               // valid_segments after one more write
static int find_device_index_(const char *device_id);                                       // devices[] index by the full id
static int build_device_segment_(int device_index, uint8_t metric_count, uint32_t saved_at_epoch, DeviceSegment *seg);
static int load_device_meta_(nvs_handle_t handle, const char *device_id, DeviceHistoryMeta *meta);
static bool save_device_meta_(nvs_handle_t handle, const char *device_id, const DeviceHistoryMeta &meta);

```

Their contract:
- `dev_ring_next_` returns `slot + 1`, or 0 when `slot + 1` is not below `max_slots` (so 0 for a slot outside the ring and for `max_slots` 0). `dev_ring_valid_` returns `valid + 1`, and never more than `max_slots`. Both are `constexpr`: the supplied test T3 runs them at compile time.
- `find_device_index_` returns the index `i` of `devices[0 .. NUM_DEVICES - 1]` whose `id` equals `device_id` in full (`strcmp`), and -1 for a null id or an id that no device has. A persist index is never used as a `devices[]` index (PR #246, review A Q4).
- `build_device_segment_` fills `*seg` for the device: first `*seg = DeviceSegment();`, then `header.metric_count = metric_count`, `header.points_per_segment = PERSIST_POINTS_PER_SEGMENT`, `header.saved_at_epoch = saved_at_epoch`. Row k (k from 0) takes the k-th metric of the device, in the order of `metric_states`, whose `history` pointer is not null (PR #246, review A Q3): `counts[k]` and `data[k]` come from `export_latest_entries_(*history, data[k], PERSIST_POINTS_PER_SEGMENT)`, the latest points, oldest first. `header.first_epoch` is the smallest non-zero epoch of `data[k][0]` over the rows, `header.last_epoch` the largest epoch of the last point of a row. It returns the number of points over all rows (0: nothing to persist), or -1 for a null `seg`, an index outside `devices[]`, a `metric_count` of 0 or above `MAX_PERSIST_METRICS`, or a device with fewer metrics with a history buffer than `metric_count`.
- `load_device_meta_` reads the blob of key `device_meta_key_(device_id)` with one `nvs_get_blob` into a `DeviceHistoryMeta` and its size. It returns 1, and fills `*meta`, only when the call succeeds, the blob has exactly `sizeof(DeviceHistoryMeta)` bytes and holds `DEV_HIST_MAGIC` and `DEV_HIST_VERSION`. It returns 0 when the key does not exist (`ESP_ERR_NVS_NOT_FOUND`) or the blob has another format (`ESP_ERR_NVS_INVALID_LENGTH`, another length, another magic or version). It returns -1, and logs with `ESP_LOGE`, when the call fails with any other error, and for a null `meta` or an empty key: a failed read of a meta that exists never starts a fresh ring (GPT Astra's audit, K1; PLAN-047). It leaves `*meta` unchanged unless it returns 1, and it never writes.
- `save_device_meta_` writes `meta` as the blob of key `device_meta_key_(device_id)`, then calls `nvs_commit`; it returns false and logs with `ESP_LOGE` when either fails.
- `persist_device_segment_` handles the persistent device `PERSIST_DEVICE_IDS[persist_index]`, with `PERSIST_METRIC_COUNTS[persist_index]` metrics, and returns its result, a `DevPersistResult`:
  1. `DEV_PERSIST_MISSING` when `find_device_index_` gives -1;
  2. `DEV_PERSIST_ERROR` when `build_device_segment_` gives -1; `DEV_PERSIST_EMPTY` when it gives 0 or `header.last_epoch` is 0;
  3. `DEV_PERSIST_ERROR` when `load_device_meta_` gives -1: it writes nothing; `DEV_PERSIST_COLLISION` when it gives 1 and the meta's `device_id` differs from the id (`strncmp` over `sizeof(meta.device_id)`): it writes nothing;
  4. a fresh meta when `load_device_meta_` gives 0, or the meta's `metric_count`, `points_per_segment` or `max_slots` differ from this device's count, `PERSIST_POINTS_PER_SEGMENT` and `DEV_PERSIST_PROVISIONAL_SLOTS`: `DeviceHistoryMeta()` with these three values and the id in `device_id` (at most `DEVICE_ID_MAX_LEN` characters and a NUL), so the ring starts at slot 0;
  5. `DEV_PERSIST_DEDUP` when the meta's `last_persist_epoch` equals `header.last_epoch`: it writes nothing;
  6. `DEV_PERSIST_NVS_LOW` when `nvs_get_stats(HISTORY_PARTITION_LABEL, ...)` fails or gives fewer `free_entries` than `DEV_PERSIST_MIN_FREE_ENTRIES`: it writes nothing;
  7. otherwise it writes the whole `DeviceSegment`, `sizeof(DeviceSegment)` bytes (or `sizeof(*seg)`), as the blob of key `device_segment_key_(id, slot)`, with `slot` the meta's `next_slot` (0 when it is not below `max_slots`); if that fails, `DEV_PERSIST_ERROR`, and the meta is not written;
  8. after a written segment, and only then, the meta with `last_written_slot = slot`, `next_slot = dev_ring_next_(slot, max_slots)`, `valid_segments = dev_ring_valid_(valid_segments, max_slots)` and `last_persist_epoch = header.last_epoch`, through `save_device_meta_`; `DEV_PERSIST_OK` when it succeeds, `DEV_PERSIST_ERROR` when it fails, and then it writes nothing more for the device in that cycle (internal round 3, L3).
  It takes a copy of the device's status with `dev_persist_status_get_(persist_index)`, sets its `last_result` to the result and adds 1 to `writes_ok` (OK), `writes_skipped` (DEDUP, EMPTY, NVS_LOW) or `writes_failed` (MISSING, COLLISION, ERROR). When it holds a meta of this device after the steps above (loaded or fresh), it copies that meta's `valid_segments`, `next_slot` and `last_persist_epoch` into that copy: the meta as written after step 8, and otherwise the meta as it was before the write, a fresh one included, also after a failed meta write and on NVS_LOW; after a collision and after a failed read it leaves them. It publishes the copy once, with `dev_persist_status_put_(persist_index, ...)`. The persist path and the handler run on two tasks: neither names `g_dev_persist_status` outside S2 (PLAN-047). It logs each result other than OK and DEDUP with `ESP_LOGW` and the id.
- `persist_all_devices_v2` does nothing when `PERSIST_DEVICE_COUNT` is 0. Otherwise it allocates one `DeviceSegment` with `new (std::nothrow)` and opens the history partition once with `nvs_open_from_partition(HISTORY_PARTITION_LABEL, HISTORY_NAMESPACE, NVS_READWRITE, &handle)`. It never initializes or erases the partition: it calls neither `open_history_nvs_` nor `ensure_history_nvs_ready_`, whose recovery can erase the whole partition; the old engine and the boot restore own that recovery, and a partition that they did not initialize makes the open fail (GPT Astra's re-audit, H4). If either fails, it sets every device's `last_result` to `DEV_PERSIST_ERROR` and adds 1 to each `writes_failed`, each through `dev_persist_status_get_` and `dev_persist_status_put_`, logs with `ESP_LOGE`, frees what it got, and returns.
