### Task Group 1: Add per-device persist functions to `firmware/core/nvs-persistence.h`

Add the following functions AFTER `persist_hourly_segment()` in `firmware/core/nvs-persistence.h`:

```cpp
// ── Phase 7: Per-device persist engine (v7.7.1.3+) ─────────────
// Writes per-device segments to NVS using the key scheme from v7.7.1.2.
// Runs alongside the old persist_hourly_segment() (dual-write) until
// v7.7.2.1 switchover. The old engine continues to be the restore source.

// Fixed provisional slot limit until retention budget calculator (v7.7.1.4)
static constexpr int DEV_PERSIST_PROVISIONAL_SLOTS = 360;  // ~15 days

static bool load_device_meta_(nvs_handle_t handle, const char *device_id,
                               DeviceHistoryMeta *meta) {
  char key[16];
  make_device_meta_key_(device_id, key, sizeof(key));
  size_t len = sizeof(DeviceHistoryMeta);
  esp_err_t err = nvs_get_blob(handle, key, meta, &len);
  if (err != ESP_OK) return false;
  return meta->magic == DEV_HIST_MAGIC &&
         meta->version == DEV_HIST_VERSION &&
         len == sizeof(DeviceHistoryMeta);
}

static bool save_device_meta_(nvs_handle_t handle, const char *device_id,
                               const DeviceHistoryMeta &meta) {
  char key[16];
  make_device_meta_key_(device_id, key, sizeof(key));
  esp_err_t err = nvs_set_blob(handle, key, &meta, sizeof(DeviceHistoryMeta));
  if (err != ESP_OK) {
    ESP_LOGE(TAG, "Failed to save device meta for %s: %s", device_id, esp_err_to_name(err));
    return false;
  }
  return true;
}

static bool persist_device_segment_(nvs_handle_t handle,
                                     const SensorEntity &device,
                                     uint32_t saved_at_epoch) {
  // Only persist devices with history-enabled metrics
  int persist_metric_count = 0;
  for (int m = 0; m < device.metric_count && m < MAX_PERSIST_METRICS; m++) {
    if (device.metric_defs[m].history_enabled &&
        device.metric_states[m].history != nullptr) {
      persist_metric_count++;
    }
  }
  if (persist_metric_count == 0) return true;  // nothing to persist — not an error

  // Load or initialize device meta
  DeviceHistoryMeta meta = {};
  bool have_meta = load_device_meta_(handle, device.id, &meta);
  if (!have_meta) {
    // Initialize fresh meta
    meta.magic = DEV_HIST_MAGIC;
    meta.version = DEV_HIST_VERSION;
    meta.metric_count = persist_metric_count;
    meta.points_per_segment = PERSIST_POINTS_PER_SEGMENT;
    meta.max_slots = DEV_PERSIST_PROVISIONAL_SLOTS;
    meta.valid_segments = 0;
    meta.next_slot = 0;
    meta.last_written_slot = 0xFFFF;
    meta.last_persist_epoch = 0;
    strncpy(meta.device_id, device.id, sizeof(meta.device_id) - 1);
    meta.device_id[sizeof(meta.device_id) - 1] = '\0';
  }

  // Dedup guard: skip if last persist was within the same segment window
  if (saved_at_epoch > 0 && meta.last_persist_epoch > 0 &&
      (saved_at_epoch - meta.last_persist_epoch) < (PERSIST_SEGMENT_HOURS * 3600 / 2)) {
    return true;  // too soon — skip
  }

  // Build DeviceSegment
  DeviceSegment segment = {};
  segment.header.magic = DEV_HIST_MAGIC;
  segment.header.version = DEV_HIST_VERSION;
  segment.header.metric_count = persist_metric_count;
  segment.header.points_per_segment = PERSIST_POINTS_PER_SEGMENT;
  segment.header.saved_at_epoch = saved_at_epoch;

  int persist_idx = 0;
  bool any_data = false;
  for (int m = 0; m < device.metric_count && m < MAX_PERSIST_METRICS; m++) {
    if (!device.metric_defs[m].history_enabled ||
        device.metric_states[m].history == nullptr) continue;

    HistoryBuffer *buf = device.metric_states[m].history;
    int exported = export_latest_entries_(*buf, segment.data[persist_idx],
                                          PERSIST_POINTS_PER_SEGMENT);
    segment.counts[persist_idx] = exported;

    if (exported > 0) {
      any_data = true;
      // Track epoch range
      uint32_t first_ep = segment.data[persist_idx][0].epoch;
      uint32_t last_ep = segment.data[persist_idx][exported - 1].epoch;
      if (segment.header.first_epoch == 0 || first_ep < segment.header.first_epoch)
        segment.header.first_epoch = first_ep;
      if (last_ep > segment.header.last_epoch)
        segment.header.last_epoch = last_ep;
    }
    persist_idx++;
  }

  if (!any_data) return true;  // no data to persist — not an error

  // Write segment to NVS
  int slot = meta.next_slot % meta.max_slots;
  char key[16];
  make_device_segment_key_(device.id, slot, key, sizeof(key));
  esp_err_t err = nvs_set_blob(handle, key, &segment, sizeof(DeviceSegment));
  if (err != ESP_OK) {
    ESP_LOGE(TAG, "Failed to write device segment %s slot %d: %s",
             device.id, slot, esp_err_to_name(err));
    return false;
  }

  // Update meta
  meta.last_written_slot = slot;
  meta.next_slot = (slot + 1) % meta.max_slots;
  if (meta.valid_segments < meta.max_slots) meta.valid_segments++;
  meta.last_persist_epoch = saved_at_epoch;

  if (!save_device_meta_(handle, device.id, meta)) return false;

  ESP_LOGI(TAG, "Persisted device %s slot %d (%d metrics, epoch %u)",
           device.id, slot, persist_metric_count, saved_at_epoch);
  return true;
}

static void persist_all_devices_v2(uint32_t saved_at_epoch = 0) {
  if (saved_at_epoch == 0) saved_at_epoch = ::time(nullptr);

  nvs_handle_t handle;
  if (!open_history_nvs_(&handle, NVS_READWRITE)) {
    ESP_LOGE(TAG, "V2 persist: failed to open NVS");
    return;
  }

  int persisted = 0;
  for (int d = 0; d < NUM_DEVICES; d++) {
    maybe_yield_nvs_scan_(d);  // Rule 61: yield between device persists
    if (persist_device_segment_(handle, devices[d], saved_at_epoch)) {
      persisted++;
    }
  }

  esp_err_t commit_err = nvs_commit(handle);
  nvs_close(handle);

  if (commit_err != ESP_OK) {
    ESP_LOGE(TAG, "V2 persist NVS commit failed: %s", esp_err_to_name(commit_err));
  } else {
    ESP_LOGI(TAG, "V2 persist complete: %d/%d devices", persisted, NUM_DEVICES);
  }
}
```
