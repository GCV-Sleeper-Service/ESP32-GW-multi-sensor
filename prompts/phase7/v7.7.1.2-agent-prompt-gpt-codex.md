// Phase 7 — Per-Device Persistence Model (v7.7.1.2+)
//
// Replaces the monolithic SegmentSnapshot model with per-device
// segments. Each device's history is stored in its own NVS key
// namespace. Adding/removing a device never touches another device's
// data. See Docs/v7.7-v7.8-persistence-architecture.md §5-6.
//
// These structs are DEFINED here but NOT WIRED into runtime code
// until v7.7.2.1 (switchover step). The old SegmentSnapshot engine
// continues to operate alongside until migration is proven.
// ═══════════════════════════════════════════════════════════════════

// ── Per-device constants ────────────────────────────────────────
static constexpr uint32_t DEV_HIST_MAGIC = 0x44485632UL;   // "DHV2"
static constexpr uint16_t DEV_HIST_VERSION = 1;
static constexpr int MAX_PERSIST_METRICS = 6;
// PERSIST_POINTS_PER_SEGMENT (=4) is already defined above for the old engine
// and is reused by the new engine — same 1-hour, 4-point segment granularity.

// ── Per-device meta (one blob per persistent device) ────────────
struct DeviceHistoryMeta {
  uint32_t magic = DEV_HIST_MAGIC;
  uint16_t version = DEV_HIST_VERSION;
  uint16_t metric_count;            // how many metrics this device persists
  uint16_t points_per_segment;      // data points per hourly segment (typically 4)
  uint16_t max_slots;               // retention capacity (budget-calculated at boot)
  uint16_t valid_segments;          // how many slots contain valid data
  uint16_t next_slot;               // ring buffer write cursor
  uint16_t last_written_slot = 0xFFFF;  // most recently written slot (dedup/debug)
  uint32_t last_persist_epoch;      // epoch of most recent persist (dedup guard)
  char     device_id[16];           // null-terminated device ID string
};

// ── EventLog for binary sensors (Phase 7, Rule 67) ─────────────
// Binary sensors use state-change-only deduplication, not periodic
// HistoryBuffer readings. See Docs/phase-V-capacity-study.md §6.
struct EventEntry {
  uint32_t epoch;   // when the state changed
  uint8_t  state;   // new state (0 or 1)
  uint8_t  _pad[3]; // padding for alignment — same wire layout as HistEntry (convenient)
};

class EventLog {
 public:
  static constexpr int CAP = 20;  // 20 events = 10 on/off cycles

  void clear() {
    std::memset(events_, 0, sizeof(events_));
    head_ = 0;
    count_ = 0;
  }

  void add(uint32_t epoch, uint8_t state) {
    events_[head_] = {epoch, state, {0, 0, 0}};
    head_ = (head_ + 1) % CAP;
    if (count_ < CAP) count_++;
  }

  int count() const { return count_; }

  EventEntry at_logical(int logical_index) const {
    if (logical_index < 0 || logical_index >= count_) return {0, 0, {0, 0, 0}};
    int start = (count_ < CAP) ? 0 : head_;
    int idx = (start + logical_index) % CAP;
    return events_[idx];
  }

  uint32_t newest_epoch() const {
    if (count_ <= 0) return 0;
    return at_logical(count_ - 1).epoch;
  }

  int export_entries(EventEntry *out, int max_entries) const {
    if (out == nullptr || max_entries <= 0) return 0;
    int n = (count_ < max_entries) ? count_ : max_entries;
    for (int i = 0; i < n; i++) out[i] = at_logical(i);
    return n;
  }

  void load_from(const EventEntry *in, int n) {
    clear();
    if (in == nullptr || n <= 0) return;
    int limit = (n < CAP) ? n : CAP;
    for (int i = 0; i < limit; i++) {
      if (in[i].epoch > 0) add(in[i].epoch, in[i].state);
    }
  }

 private:
  EventEntry events_[CAP] = {};
  int head_ = 0;
  int count_ = 0;
};
// EventLog uses ~78% less RAM than HistoryBuffer (state-change-only vs periodic sampling)

### Task Group 2: Add FNV-1a hash and key helpers to `firmware/core/nvs-persistence.h`

Add the following block to `firmware/core/nvs-persistence.h` AFTER the existing `make_segment_key_()` function (line ~57) and BEFORE `clear_runtime_histories_()` (line ~58). This positions the new key helpers near the existing key-generation code.

```cpp
// ── Phase 7: Per-device NVS key helpers (v7.7.1.2+) ────────────
// FNV-1a 32-bit hash of device ID. Produces an 8-hex-char hash
// that fits within the 15-char NVS key limit alongside prefixes
// and slot numbers. See Docs/v7.7-v7.8-persistence-architecture.md §6.

static uint32_t device_id_hash_(const char *id) {
  uint32_t h = 0x811c9dc5;
  for (const char *p = id; *p; p++) {
    h ^= (uint8_t)*p;
    h *= 0x01000193;
  }
  return h;
}

// Key format: "dm_HHHHHHHH" (3 + 8 = 11 chars, well within 15-char limit)
static void make_device_meta_key_(const char *device_id, char *key, size_t key_len) {
  uint32_t hash = device_id_hash_(device_id);
  snprintf(key, key_len, "dm_%08x", hash);
}

// Key format: "ds_HHHHHHHH_NNN" (3 + 8 + 1 + 3 = 15 chars, exactly at limit)
static void make_device_segment_key_(const char *device_id, int slot, char *key, size_t key_len) {
  uint32_t hash = device_id_hash_(device_id);
  snprintf(key, key_len, "ds_%08x_%03d", hash, slot);
}
```
