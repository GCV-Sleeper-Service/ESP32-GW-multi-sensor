## 5. Exact Scope — Dashboard Add/Remove/Test UI

Modify `renderSettingsPanel()` in `dashboard.js` to replace the read-only satellite list with interactive controls. **NO firmware changes in this step. All changes are JS + HTML + dashboard.h regeneration.**

### 5a. UI components to add

The Settings panel gets three new elements:

**1. Add Satellite form** — placed at the top of the settings panel:
- URL text input (`<input type="text" placeholder="http://192.168.x.x">`)
- Optional name text input (`<input type="text" placeholder="Friendly name (optional)">`)
- "Test" button — calls `POST /api/aggregator/test-satellite?url=...`
- "Add" button — calls `POST /api/aggregator/add-satellite?url=...&name=...`
- Status area — shows test results or error messages inline (not `alert()`)

**2. Per-satellite remove button** — on each satellite card:
- "Remove" button
- Click triggers `confirm()` dialog: "Remove satellite {name}? This cannot be undone."
- On confirm: calls `DELETE /api/aggregator/satellite/{id}`
- On success: re-fetches gateway list and re-renders

**3. Enhanced status display** per satellite:
- Last-seen timestamp (from `gw.last_seen` epoch — format as relative time or absolute)
- Consecutive failure count if > 0 (show as warning indicator)

### 5b. Data flow trace: Add Satellite

```
User enters URL → clicks "Test"
  → fetch POST /api/aggregator/test-satellite?url=...
  → Display result inline:
    - Success: "✓ Found: {name} ({hardware}, {sensor_count} devices)"
    - Failure: "✗ {error message}"

User clicks "Add"
  → fetch POST /api/aggregator/add-satellite?url=...&name=...
  → On 200: clear form, re-fetch gateways, re-render settings panel
  → On 400/409: display error message inline
```

### 5c. Data flow trace: Remove Satellite

```
User clicks "Remove" on satellite card
  → confirm("Remove satellite {name}?")
  → If confirmed: fetch DELETE /api/aggregator/satellite/{id}
  → On 200: re-fetch gateways, re-render settings panel
  → On 404: display "Satellite not found" inline (race condition — already removed)
```

### 5d. Implementation — `renderSettingsPanel()` replacement

Replace the ENTIRE body of `renderSettingsPanel(gateways)` with the new interactive version. Key requirements:

**HTML escaping:** Use `escHtml()` on ALL dynamic text:
- `escHtml(gw.name)` — satellite names
- `escHtml(gw.id)` — satellite IDs
- `escHtml(gw.base_url || '—')` — URLs
- `escHtml(fwVersion)` — firmware versions
- Any text from test-satellite response

**Event binding:** Use programmatic event binding. Do NOT use inline `onclick`, `oninput`, or `onchange` attributes in the HTML string. Instead:
```javascript
// ✅ Correct — programmatic binding after innerHTML set
var btn = document.getElementById('add-sat-btn');
if (btn) btn.addEventListener('click', handleAddSatellite);

// ❌ PROHIBITED — inline event handler
html += '<button onclick="handleAddSatellite()">Add</button>';
```

**Null checks:** Use explicit checks (`!== undefined && !== null`), never truthy checks on values that could be `0` (like `last_seen` epoch or `consecutive_failures`).

**isFinite guard:** Not applicable for this step (no numeric-to-CSS conversions expected). But if you add any progress bars or width calculations, verify `isFinite()` before setting CSS values.

**Dark/light mode (LESSON-OPS-065):** All new `<input>` and `<button>` elements need:
```css
color-scheme: light dark;
```
This is applied via the existing `.settings-panel input, .settings-panel button` selector. If no such selector exists, add one to the `<style>` section in dashboard.html AND in the inline styles in dashboard.js.

### 5e. Skeleton implementation

```javascript
function renderSettingsPanel(gateways) {
  var grid = document.getElementById('gwGrid');
  grid.innerHTML = '';

  var html = '<div class="settings-panel">';
  html += '<h3 class="settings-title">Satellite Management</h3>';

  // ── Add Satellite form ──
  html += '<div class="settings-add-form" id="add-sat-form">';
  html += '<div class="settings-add-row">';
  html += '<input type="text" id="add-sat-url" placeholder="http://192.168.x.x" class="settings-input" />';
  html += '<input type="text" id="add-sat-name" placeholder="Name (optional)" class="settings-input settings-input-name" />';
  html += '</div>';
  html += '<div class="settings-add-actions">';
  html += '<button id="test-sat-btn" class="settings-btn settings-btn-test">Test</button>';
  html += '<button id="add-sat-btn" class="settings-btn settings-btn-add">Add</button>';
  html += '</div>';
  html += '<div id="add-sat-status" class="settings-status"></div>';
  html += '</div>';

  // ── Satellite list ──
  if (!gateways || gateways.length === 0) {
    html += '<p class="settings-empty">No satellites configured.</p>';
  } else {
    gateways.forEach(function(gw) {
      var statusDot = gw.reachable ? '\uD83D\uDFE2' : '\uD83D\uDD34';
      var fwVersion = '\u2014';
      var deviceCount = '\u2014';
      if (gw.firmware_version) fwVersion = gw.firmware_version;
      if (gw.device_count !== undefined && gw.device_count !== null) deviceCount = gw.device_count;
      else if (gw.sensor_count !== undefined && gw.sensor_count !== null) deviceCount = gw.sensor_count;

      html += '<div class="settings-satellite-card">';
      html += '<div class="settings-sat-header">' + statusDot + ' ' + escHtml(gw.name) +
        ' <span class="settings-sat-id">(' + escHtml(gw.id) + ')</span></div>';
      html += '<div class="settings-sat-details">';
      html += '<div>URL: <code>' + escHtml(gw.base_url || '\u2014') + '</code></div>';
      html += '<div>Firmware: ' + escHtml(fwVersion) + '</div>';
      html += '<div>Devices: ' + escHtml(String(deviceCount)) + '</div>';
      html += '<div>Status: ' + (gw.reachable ? 'Online' : 'Unreachable') + '</div>';

      // Last seen
      if (gw.last_seen !== undefined && gw.last_seen !== null && gw.last_seen > 0) {
        var seenDate = new Date(gw.last_seen * 1000);
        html += '<div>Last seen: ' + escHtml(seenDate.toLocaleString()) + '</div>';
      }

      // Consecutive failures warning
      if (gw.consecutive_failures !== undefined && gw.consecutive_failures !== null && gw.consecutive_failures > 0) {
        html += '<div class="settings-warning">\u26A0 ' + gw.consecutive_failures + ' consecutive failure(s)</div>';
      }

      html += '</div>';
      // Remove button — data-id attribute stores the satellite ID
      html += '<button class="settings-btn settings-btn-remove" data-sat-id="' + escHtml(gw.id) +
        '" data-sat-name="' + escHtml(gw.name) + '">Remove</button>';
      html += '</div>';
    });
  }

  html += '</div>';
  grid.innerHTML = html;

  // ── Bind event handlers (programmatic — no inline onclick) ──
  var testBtn = document.getElementById('test-sat-btn');
  if (testBtn) testBtn.addEventListener('click', _handleTestSatellite);

  var addBtn = document.getElementById('add-sat-btn');
  if (addBtn) addBtn.addEventListener('click', _handleAddSatellite);

  var removeBtns = document.querySelectorAll('.settings-btn-remove');
  removeBtns.forEach(function(btn) {
    btn.addEventListener('click', function() {
      var satId = this.getAttribute('data-sat-id');
      var satName = this.getAttribute('data-sat-name');
      _handleRemoveSatellite(satId, satName);
    });
  });
}
```

### 5f. Handler functions

Add these functions near `renderSettingsPanel()`:

```javascript
function _handleTestSatellite() {
  var urlInput = document.getElementById('add-sat-url');
  var statusDiv = document.getElementById('add-sat-status');
  if (!urlInput || !statusDiv) return;
  var url = urlInput.value.trim();
  if (!url) { statusDiv.textContent = 'Enter a URL first.'; return; }

  statusDiv.textContent = 'Testing...';
  fetch(ESP_HOST + '/api/aggregator/test-satellite?url=' + encodeURIComponent(url), {
    method: 'POST', cache: 'no-store'
  }).then(function(r) { return r.json(); })
    .then(function(data) {
      if (data.ok && data.gateway) {
        var gw = data.gateway;
        statusDiv.innerHTML = '\u2713 Found: <strong>' + escHtml(gw.name || gw.id) +
          '</strong> (' + escHtml(gw.hardware || '?') + ', ' +
          (gw.sensor_count !== undefined && gw.sensor_count !== null ? gw.sensor_count : '?') +
          ' devices)';
      } else {
        statusDiv.textContent = '\u2717 ' + (data.message || 'Test failed');
      }
    })
    .catch(function(err) {
      statusDiv.textContent = '\u2717 Connection error: ' + err.message;
    });
}

function _handleAddSatellite() {
  var urlInput = document.getElementById('add-sat-url');
  var nameInput = document.getElementById('add-sat-name');
  var statusDiv = document.getElementById('add-sat-status');
  if (!urlInput || !statusDiv) return;
  var url = urlInput.value.trim();
  if (!url) { statusDiv.textContent = 'Enter a URL first.'; return; }

  var params = 'url=' + encodeURIComponent(url);
  if (nameInput && nameInput.value.trim()) {
    params += '&name=' + encodeURIComponent(nameInput.value.trim());
  }

  statusDiv.textContent = 'Adding...';
  fetch(ESP_HOST + '/api/aggregator/add-satellite?' + params, {
    method: 'POST', cache: 'no-store'
  }).then(function(r) { return r.json(); })
    .then(function(data) {
      if (data.ok) {
        statusDiv.textContent = '\u2713 Added successfully';
        if (urlInput) urlInput.value = '';
        if (nameInput) nameInput.value = '';
        // Re-fetch and re-render
        _refreshSettingsPanel();
      } else {
        statusDiv.textContent = '\u2717 ' + (data.message || 'Add failed');
      }
    })
    .catch(function(err) {
      statusDiv.textContent = '\u2717 Connection error: ' + err.message;
    });
}

function _handleRemoveSatellite(satId, satName) {
  if (!confirm('Remove satellite ' + satName + '? This cannot be undone.')) return;

  fetch(ESP_HOST + '/api/aggregator/satellite/' + encodeURIComponent(satId), {
    method: 'DELETE', cache: 'no-store'
  }).then(function(r) { return r.json(); })
    .then(function(data) {
      if (data.ok) {
        _refreshSettingsPanel();
      } else {
        var statusDiv = document.getElementById('add-sat-status');
        if (statusDiv) statusDiv.textContent = '\u2717 ' + (data.message || 'Remove failed');
      }
    })
    .catch(function(err) {
      var statusDiv = document.getElementById('add-sat-status');
      if (statusDiv) statusDiv.textContent = '\u2717 Connection error: ' + err.message;
    });
}

function _refreshSettingsPanel() {
  fetch(ESP_HOST + '/api/aggregator/gateways', {cache: 'no-store'})
    .then(function(r) { return r.json(); })
    .then(function(data) {
      if (data.gateways) {
        window._aggregatorGateways = data.gateways;
        renderSettingsPanel(data.gateways);
      }
    })
    .catch(function() { /* silent — next periodic poll will refresh */ });
}
```
