### 5b. Implement `buildSystemCard()`

The system card shows CPU, RAM, and disk as horizontal usage bars (0-100%), plus an uptime value. Usage bars have color coding: green below 60%, yellow 60-80%, red above 80%.

```js
function buildSystemCard(s, manifest) {
  var description = '';
  if (manifest && manifest.sensors) {
    var entry = manifest.sensors.find(function(ms) { return ms.id === s.id; });
    if (entry && entry.source && entry.source.description) description = entry.source.description;
  }
  return (
    '<div class="sensor-card system-card" data-device-id="' + s.id + '">' +
      '<div class="sensor-card-header">' +
        '<input class="sensor-color-picker" id="picker-' + s.id + '" type="color" value="' + (s.color || '#66BB6A') + '" data-sensor-color="' + s.id + '">' +
        s.name +
      '</div>' +
      '<div class="sensor-readings">' +
        buildUsageBarRow('CPU', 'sys-cpu-' + s.id) +
        buildUsageBarRow('RAM', 'sys-ram-' + s.id) +
        buildUsageBarRow('Disk', 'sys-disk-' + s.id) +
        '<div class="sensor-reading">' +
          '<div class="reading-label">Uptime</div>' +
          '<div class="reading-value waiting" id="sys-uptime-' + s.id + '">&mdash;</div>' +
        '</div>' +
        '<div class="sensor-reading">' +
          '<div class="reading-label">Last Seen</div>' +
          '<div class="reading-time" id="sys-lastseen-' + s.id + '">last: &mdash;</div>' +
        '</div>' +
        (description ? '<div class="sensor-reading"><div class="reading-label" style="font-size:.52rem;color:var(--text-muted)">' + description + '</div></div>' : '') +
      '</div>' +
    '</div>'
  );
}

function buildUsageBarRow(label, id) {
  return (
    '<div class="sensor-reading system-usage-row">' +
      '<div class="reading-label">' + label + '</div>' +
      '<div class="system-bar-bg"><div class="system-bar-fill" id="bar-' + id + '"></div></div>' +
      '<div class="reading-value waiting" id="val-' + id + '">&mdash;</div>' +
    '</div>'
  );
}
```

### 5c. Add metric formatters

```js
METRIC_FORMATTERS.cpu_usage = function(value) {
  if (value === null || value === undefined || isNaN(value)) return '—';
  return value.toFixed(1) + '%';
};
METRIC_FORMATTERS.ram_usage = function(value) {
  if (value === null || value === undefined || isNaN(value)) return '—';
  return value.toFixed(1) + '%';
};
METRIC_FORMATTERS.disk_usage = function(value) {
  if (value === null || value === undefined || isNaN(value)) return '—';
  return value.toFixed(1) + '%';
};
METRIC_FORMATTERS.uptime_hours = function(value) {
  if (value === null || value === undefined || isNaN(value)) return '—';
  if (value >= 24) return (value / 24).toFixed(1) + ' days';
  return value.toFixed(1) + ' hrs';
};
```

### 5d. ⚠️ CRITICAL: System card data path — `/api/v2/live` polling

System device data does NOT come through ESPHome SSE state events (same as network devices). Data comes from `/api/v2/live` polling. The existing `pollV2Live()` fetches this endpoint every 15 seconds.

Extend `updateNetworkCards()` (or create `updateSystemCards()`) to also handle system devices:

```js
function updateSystemCards(liveData) {
  if (!liveData || !liveData.devices) return;
  SENSORS.forEach(function(s) {
    if (s.category !== 'system') return;
    var devData = liveData.devices[s.id];
    if (!devData) return;
    
    // CPU
    updateUsageBar('sys-cpu-' + s.id, devData.cpu_pct, METRIC_FORMATTERS.cpu_usage);
    // RAM
    updateUsageBar('sys-ram-' + s.id, devData.ram_pct, METRIC_FORMATTERS.ram_usage);
    // Disk
    updateUsageBar('sys-disk-' + s.id, devData.disk_pct, METRIC_FORMATTERS.disk_usage);
    // Uptime
    var uptimeEl = document.getElementById('sys-uptime-' + s.id);
    if (uptimeEl && devData.uptime_hrs !== undefined && devData.uptime_hrs !== null) {
      uptimeEl.textContent = METRIC_FORMATTERS.uptime_hours(devData.uptime_hrs);
      uptimeEl.classList.remove('waiting');
    }
    // Last seen
    var seenEl = document.getElementById('sys-lastseen-' + s.id);
    if (seenEl && devData.last_seen) {
      seenEl.textContent = 'last: ' + new Date(devData.last_seen * 1000).toLocaleTimeString([], {hour12:false});
    }
  });
}

function updateUsageBar(id, value, formatter) {
  var barEl = document.getElementById('bar-' + id);
  var valEl = document.getElementById('val-' + id);
  if (value === null || value === undefined) return;
  var pct = Math.min(100, Math.max(0, value));
  if (barEl) {
    barEl.style.width = pct + '%';
    barEl.className = 'system-bar-fill' +
      (pct >= 80 ? ' bar-danger' : pct >= 60 ? ' bar-warning' : ' bar-ok');
  }
  if (valEl) {
    valEl.textContent = formatter(value);
    valEl.classList.remove('waiting');
  }
}
```

Wire into the existing `pollV2Live()` response handler:
```js
.then(function(data) { updateNetworkCards(data); updateSystemCards(data); })
```
