## 5. Exact Scope — Aggregator Dashboard UI

Extend the dashboard with multi-gateway UI for aggregator mode. Satellite-mode dashboard must be completely unaffected.

### 5a. Aggregator mode detection — runtime, not compile-time

The dashboard detects aggregator mode at runtime by probing the API:

```js
var DASHBOARD_MODE = 'satellite'; // default

async function detectAggregatorMode() {
  try {
    var r = await fetch(ESP_HOST + '/api/aggregator/gateways', {cache: 'no-store'});
    if (r.ok) {
      var data = await r.json();
      if (data && Array.isArray(data.gateways) && data.gateways.length > 0) {
        DASHBOARD_MODE = 'aggregator';
        window._aggregatorGateways = data.gateways;
        return true;
      }
    }
  } catch(e) { /* not aggregator */ }
  return false;
}
```

**Why runtime, not compile-time:** The same `dashboard.html` is embedded in both satellite and aggregator firmware. The dashboard must work on both. The API probe determines which mode to activate.

### 5b. Modified boot flow

The existing boot flow is:
1. `loadManifestV2()` → fetch `/api/manifest`
2. `applySensorMeta()` → build SENSORS array
3. `buildDeviceCards()` → render cards
4. `connectSSE()` / start polling

For aggregator mode, insert aggregator detection BEFORE the manifest load:

```js
async function initDashboard() {
  // ... existing init ...
  
  var isAggregator = await detectAggregatorMode();
  
  if (isAggregator) {
    await initAggregatorDashboard();
  } else {
    // Existing satellite boot flow — UNCHANGED
    await loadManifestV2();
    buildDeviceCards();
    connectSSE();
    // ... rest of satellite init
  }
}
```

**Critical:** If `detectAggregatorMode()` returns false (satellite), the ENTIRE aggregator code path is skipped. No new DOM elements, no new fetch calls, no CSS changes applied.

async function initAggregatorDashboard() {

### 5d. Gateway selector bar

Add a horizontal tab bar above the sensor grid:

```js
function renderGatewaySelector(gateways) {
  var container = document.getElementById('sensorGrid');
  var selectorHtml = '<div id="gwSelector" class="gw-selector">' +
    '<button class="gw-tab active" data-gw="all">All Gateways</button>';
  
  gateways.forEach(function(gw) {
    var statusClass = gw.reachable ? 'gw-online' : 'gw-offline';
    selectorHtml += '<button class="gw-tab ' + statusClass + '" data-gw="' + gw.id + '">' +
      gw.name + '</button>';
  });
  
  selectorHtml += '</div>';
  container.insertAdjacentHTML('beforebegin', selectorHtml);
  
  // Bind click handlers
  document.querySelectorAll('.gw-tab').forEach(function(tab) {
    tab.addEventListener('click', function() {
      document.querySelectorAll('.gw-tab').forEach(function(t) { t.classList.remove('active'); });
      tab.classList.add('active');
      var gwId = tab.getAttribute('data-gw');
      if (gwId === 'all') {
        renderAllGatewaysSummary(window._aggregatorGateways);
      } else {
        renderGatewayDevices(gwId);
      }
    });
  });
}
```

### 5f. Per-gateway device view

When a specific gateway tab is clicked:

1. Fetch that gateway's cached manifest from `/api/aggregator/gateways` (already cached in `window._aggregatorGateways`)
2. Parse the manifest to extract devices
3. Build sensor configs with namespaced IDs: `{gw_id}.{device_id}` (e.g., `gw-main.office`)
4. Render cards using existing `CARD_RENDERERS`
5. Fetch live values from `/api/aggregator/live` to populate cards

```js
function renderGatewayDevices(gwId) {
  var grid = document.getElementById('sensorGrid');
  grid.innerHTML = '<div class="gw-loading">Loading devices for ' + gwId + '...</div>';
  
  // Fetch aggregator live data to get device values
  fetch(ESP_HOST + '/api/aggregator/live', {cache: 'no-store'})
    .then(function(r) { return r.json(); })
    .then(function(liveData) {
      var gwData = liveData.gateways && liveData.gateways[gwId];
      if (!gwData) {
        grid.innerHTML = '<div class="gw-error">Gateway ' + gwId + ' data not available</div>';
        return;
      }
      
      // Parse cached manifest for this gateway to get device list
      var gwInfo = window._aggregatorGateways.find(function(g) { return g.id === gwId; });
      if (!gwInfo || !gwInfo.manifest_cached) {
        grid.innerHTML = '<div class="gw-error">No manifest for ' + gwId + '</div>';
        return;
      }
      
      // The manifest is cached as raw JSON in the gateways response
      // You may need to enhance /api/aggregator/gateways to include parsed manifest
      // OR fetch the manifest separately from the aggregator cache
      
      // For now, render a basic view from the live data
      grid.innerHTML = '';
      if (gwData.live) {
        try {
          var live = typeof gwData.live === 'string' ? JSON.parse(gwData.live) : gwData.live;
          if (live.devices) {
            Object.keys(live.devices).forEach(function(deviceId) {
              // Use _default renderer for now — full manifest-driven rendering
              // requires passing the parsed manifest through the gateway endpoint
              var device = { id: gwId + '.' + deviceId, name: deviceId };
              var renderer = CARD_RENDERERS._default;
              grid.innerHTML += renderer(device, null);
            });
          }
        } catch(e) {
          grid.innerHTML = '<div class="gw-error">Error parsing live data</div>';
        }
      }
    })
    .catch(function(err) {
      grid.innerHTML = '<div class="gw-error">Failed to load: ' + err.message + '</div>';
    });
}
```

**Important architectural note:** For full manifest-driven rendering in the per-gateway view, the `/api/aggregator/gateways` endpoint (from v7.5.5.2) should include the cached manifest JSON for each gateway. If it doesn't currently, extend it to include a `manifest` field. This lets the dashboard parse device types and use the correct `CARD_RENDERERS` per category.

### 5g. Aggregator live polling

```js
var _aggLiveInFlight = false;
function pollAggregatorLive() {
  if (_aggLiveInFlight) return;
  _aggLiveInFlight = true;
  
  fetch(ESP_HOST + '/api/aggregator/gateways', {cache: 'no-store'})
    .then(function(r) { return r.json(); })
    .then(function(data) {
      if (data && data.gateways) {
        window._aggregatorGateways = data.gateways;
        // If "All Gateways" view is active, refresh summary
        var activeTab = document.querySelector('.gw-tab.active');
        if (activeTab && activeTab.getAttribute('data-gw') === 'all') {
          renderAllGatewaysSummary(data.gateways);
        }
      }
    })
    .catch(function() { /* silent — aggregator may be restarting */ })
    .finally(function() { _aggLiveInFlight = false; });
}
```

### 5h. History charts through proxy

When a user views a specific gateway's devices and opens a chart, history must be fetched through the aggregator proxy, not directly from the satellite. Override the history URL resolution:

For aggregator mode, when building the history fetch URL, prefix with the proxy path:
`/api/aggregator/proxy/{gw_id}/history/{device}/{metric}` instead of `/history/{device}/{metric}` or `/api/v2/history/{device}/{metric}`

This can be done by setting a per-device `history_base_url` that `fetchDeviceHistory()` uses.
